    <script>
        let factorA = 0;
        let factorB = 0;
        let targetNumber = 0;
        let currentNumber = 0;
        let xp = 0;
        const maxXP = 100;
        
        let counts = {
            100: 0,
            10: 0,
            1: 0
        };

        // Theme data for working memory variety
        const themes = [
            { animal: '🐒', animalName: 'Mischievous Monkeys', container: 'Crates', containerSingular: 'Crate', item: 'Bananas', emoji: '🍌' },
            { animal: '🐘', animalName: 'Stomping Elephants', container: 'Baskets', containerSingular: 'Basket', item: 'Peanuts', emoji: '🥜' },
            { animal: '🦒', animalName: 'Whispering Giraffes', container: 'Branches', containerSingular: 'Branch', item: 'Leaves', emoji: '🍃' },
            { animal: '🦓', animalName: 'Dazzling Zebras', container: 'Patches', containerSingular: 'Patch', item: 'Tall Grass', emoji: '🌾' }
        ];
        let currentTheme = themes[0];

        const uiElements = {
            factorA: document.getElementById('factor-a'),
            factorB: document.getElementById('factor-b'),
            labelA: document.getElementById('label-a'),
            labelB: document.getElementById('label-b'),
            storyProblem: document.getElementById('story-problem'),
            currentDisplay: document.getElementById('current-number'),
            xpFill: document.getElementById('xp-fill'),
            xpDisplay: document.getElementById('xp-display'),
            modalOverlay: document.getElementById('modal-overlay'),
            modalBox: document.getElementById('modal-box'),
            modalIcon: document.getElementById('modal-icon'),
            headerAnimal: document.getElementById('header-animal'),
            count100: document.getElementById('count-100'),
            count10: document.getElementById('count-10'),
            count1: document.getElementById('count-1')
        };

        const AudioContext = window.AudioContext || window.webkitAudioContext;
        let audioCtx;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new AudioContext();
            }
        }

        function playSoothingChime() {
            initAudio();
            if(audioCtx.state === 'suspended') audioCtx.resume();
            
            // A soothing major chord arpeggio to celebrate
            const frequencies = [261.63, 329.63, 392.00, 523.25]; 
            
            frequencies.forEach((freq, index) => {
                setTimeout(() => {
                    const osc = audioCtx.createOscillator();
                    const gainNode = audioCtx.createGain();
                    
                    osc.type = 'sine'; 
                    osc.frequency.value = freq;
                    
                    osc.connect(gainNode);
                    gainNode.connect(audioCtx.destination);
                    
                    gainNode.gain.setValueAtTime(0, audioCtx.currentTime);
                    gainNode.gain.linearRampToValueAtTime(0.3, audioCtx.currentTime + 0.1);
                    gainNode.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1.5);
                    
                    osc.start();
                    osc.stop(audioCtx.currentTime + 1.5);
                }, index * 150); 
            });
        }

        function playPopSound() {
            initAudio();
            if(audioCtx.state === 'suspended') audioCtx.resume();
            
            const osc = audioCtx.createOscillator();
            const gainNode = audioCtx.createGain();
            
            osc.type = 'triangle';
            osc.frequency.setValueAtTime(400, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(600, audioCtx.currentTime + 0.1);
            
            gainNode.gain.setValueAtTime(0.1, audioCtx.currentTime);
            gainNode.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
            
            osc.connect(gainNode);
            gainNode.connect(audioCtx.destination);
            
            osc.start();
            osc.stop(audioCtx.currentTime + 0.1);
        }

        function generateProblem() {
            // Pick a random safari theme
            currentTheme = themes[Math.floor(Math.random() * themes.length)];
            uiElements.headerAnimal.innerText = currentTheme.animal;
            uiElements.modalIcon.innerText = currentTheme.animal;

            // Difficulty Scaling based on XP
            let minA = 1, maxA = 5;
            let minB = 1, maxB = 5;
            
            if (xp >= 30 && xp < 70) { 
                // Medium difficulty
                minA = 2; maxA = 9;
                minB = 2; maxB = 5;
            } else if (xp >= 70) { 
                // Harder difficulty
                minA = 3; maxA = 9;
                minB = 3; maxB = 9;
            }

            factorA = Math.floor(Math.random() * (maxA - minA + 1)) + minA;
            factorB = Math.floor(Math.random() * (maxB - minB + 1)) + minB;
            targetNumber = factorA * factorB;

            // Update UI to externalize working memory
            uiElements.factorA.innerText = factorA;
            uiElements.factorB.innerText = factorB;
            uiElements.labelA.innerText = currentTheme.container;
            uiElements.labelB.innerText = currentTheme.item;
            
            uiElements.storyProblem.innerText = `The ${currentTheme.animalName} have ${factorA} ${currentTheme.container.toLowerCase()}. Each ${currentTheme.containerSingular.toLowerCase()} has ${factorB} ${currentTheme.item.toLowerCase()}. How many ${currentTheme.item.toLowerCase()} total?`;
        }

        function updateDisplay() {
            uiElements.currentDisplay.innerText = currentNumber;
            uiElements.count100.innerText = counts[100];
            uiElements.count10.innerText = counts[10];
            uiElements.count1.innerText = counts[1];

            // Evaluate Win Condition
            if (currentNumber === targetNumber) {
                handleWin();
            } else if (currentNumber > targetNumber) {
                // Gentle error correction - visual only, no harsh sounds
                uiElements.currentDisplay.classList.add('text-rose-500');
                uiElements.currentDisplay.classList.remove('text-amber-400');
                setTimeout(() => {
                    uiElements.currentDisplay.classList.remove('text-rose-500');
                    uiElements.currentDisplay.classList.add('text-amber-400');
                }, 500);
            }
        }

        function addValue(val) {
            playPopSound();
            currentNumber += val;
            counts[val]++;
            updateDisplay();
        }

        function resetCurrent() {
            currentNumber = 0;
            counts = { 100: 0, 10: 0, 1: 0 };
            updateDisplay();
        }

        function handleWin() {
            playSoothingChime();
            fireConfetti();
            
            // Increment XP safely
            xp += 10;
            if(xp > maxXP) xp = maxXP;
            
            uiElements.xpFill.style.width = `${(xp / maxXP) * 100}%`;
            uiElements.xpDisplay.innerText = `${xp} / ${maxXP}`;

            // Show Success Modal
            setTimeout(() => {
                uiElements.modalOverlay.classList.remove('hidden');
                uiElements.modalBox.style.transform = "scale(1)";
                
                if(xp >= maxXP) {
                    document.getElementById('modal-title').innerText = "Hooray!";
                    document.getElementById('modal-message').innerText = `You helped all the animals and mastered multiplication!`;
                } else {
                    document.getElementById('modal-title').innerText = "Wackadoo!";
                    document.getElementById('modal-message').innerText = `You correctly counted ${targetNumber} ${currentTheme.item.toLowerCase()}!`;
                }
            }, 600);
        }

        function nextLevel() {
            uiElements.modalBox.style.transform = "scale(0.95)";
            setTimeout(() => {
                uiElements.modalOverlay.classList.add('hidden');
                resetCurrent();
                
                if(xp >= maxXP) {
                    // Reset game state for endless play
                    xp = 0;
                    uiElements.xpFill.style.width = `0%`;
                    uiElements.xpDisplay.innerText = `0 / ${maxXP}`;
                }
                
                generateProblem();
            }, 200);
        }

        function fireConfetti() {
            const emojis = [currentTheme.animal, currentTheme.emoji, '✨', '💎', '🌟'];
            for (let i = 0; i < 35; i++) {
                const conf = document.createElement('div');
                conf.innerText = emojis[Math.floor(Math.random() * emojis.length)];
                conf.classList.add('confetti');
                conf.style.left = Math.random() * 100 + 'vw';
                conf.style.top = -50 + 'px';
                // Randomize animation duration for a natural fall effect
                conf.style.animationDuration = (Math.random() * 2 + 2) + 's';
                document.body.appendChild(conf);
                
                // Cleanup DOM
                setTimeout(() => {
                    conf.remove();
                }, 4000);
            }
        }

        window.onload = () => {
            // First interaction required for audio API, handled seamlessly by the play buttons
            generateProblem();
        };

    </script>

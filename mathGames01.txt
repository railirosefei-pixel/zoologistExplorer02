    <script>
        let targetNumber = 0;
        let currentNumber = 0;
        let xp = 0;
        const maxXP = 100;
        
        let counts = {
            1000: 0,
            100: 0,
            10: 0,
            1: 0
        };

        const targetDisplay = document.getElementById('target-number');
        const currentDisplay = document.getElementById('current-number');
        const xpFill = document.getElementById('xp-fill');
        const xpDisplay = document.getElementById('xp-display');
        const modalOverlay = document.getElementById('modal-overlay');
        const modalBox = document.getElementById('modal-box');

        // Audio Context for soothing sounds (avoids external files)
        const AudioContext = window.AudioContext || window.webkitAudioContext;
        let audioCtx;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new AudioContext();
            }
        }

        function playSoothingChime() {
            initAudio();
            if(audioCtx.state === 'suspended') audioCtx.resume();
            
            // A soothing major chord arpeggio (C, E, G, C)
            const frequencies = [261.63, 329.63, 392.00, 523.25]; 
            
            frequencies.forEach((freq, index) => {
                setTimeout(() => {
                    const osc = audioCtx.createOscillator();
                    const gainNode = audioCtx.createGain();
                    
                    osc.type = 'sine'; // Soft sine wave
                    osc.frequency.value = freq;
                    
                    osc.connect(gainNode);
                    gainNode.connect(audioCtx.destination);
                    
                    // Fade in and out to make it gentle
                    gainNode.gain.setValueAtTime(0, audioCtx.currentTime);
                    gainNode.gain.linearRampToValueAtTime(0.3, audioCtx.currentTime + 0.1);
                    gainNode.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1.5);
                    
                    osc.start();
                    osc.stop(audioCtx.currentTime + 1.5);
                }, index * 150); // Play notes in sequence
            });
        }

        function playPopSound() {
            initAudio();
            if(audioCtx.state === 'suspended') audioCtx.resume();
            
            const osc = audioCtx.createOscillator();
            const gainNode = audioCtx.createGain();
            
            osc.type = 'triangle';
            osc.frequency.setValueAtTime(600, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(800, audioCtx.currentTime + 0.1);
            
            gainNode.gain.setValueAtTime(0.1, audioCtx.currentTime);
            gainNode.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
            
            osc.connect(gainNode);
            gainNode.connect(audioCtx.destination);
            
            osc.start();
            osc.stop(audioCtx.currentTime + 0.1);
        }

        function generateTarget() {
            // Adjust difficulty based on XP
            let min = 10;
            let max = 99;
            
            if (xp >= 20) { min = 100; max = 999; }
            if (xp >= 50) { min = 1000; max = 9999; }

            targetNumber = Math.floor(Math.random() * (max - min + 1)) + min;
            targetDisplay.innerText = targetNumber.toLocaleString();
        }

        function updateDisplay() {
            currentDisplay.innerText = currentNumber.toLocaleString();
            document.getElementById('count-1000').innerText = counts[1000];
            document.getElementById('count-100').innerText = counts[100];
            document.getElementById('count-10').innerText = counts[10];
            document.getElementById('count-1').innerText = counts[1];

            // Check win condition
            if (currentNumber === targetNumber) {
                handleWin();
            } else if (currentNumber > targetNumber) {
                // Friendly correction without alert
                currentDisplay.style.color = "#ef4444"; // Red
                setTimeout(() => {
                    currentDisplay.style.color = "#fbbf24"; // Back to amber
                }, 500);
            }
        }

        function addValue(val) {
            playPopSound();
            currentNumber += val;
            counts[val]++;
            updateDisplay();
        }

        function resetCurrent() {
            currentNumber = 0;
            counts = { 1000: 0, 100: 0, 10: 0, 1: 0 };
            updateDisplay();
        }

        function handleWin() {
            playSoothingChime();
            fireConfetti();
            
            // Add XP
            xp += 10;
            if(xp > maxXP) xp = maxXP;
            
            xpFill.style.width = `${(xp / maxXP) * 100}%`;
            xpDisplay.innerText = `${xp} / ${maxXP}`;

            // Show Modal
            setTimeout(() => {
                modalOverlay.classList.remove('hidden');
                modalBox.style.transform = "scale(1)";
                
                if(xp >= maxXP) {
                    document.getElementById('modal-title').innerText = "Hooray!";
                    document.getElementById('modal-message').innerText = "You filled your Bestiary and saved the Savanna!";
                }
            }, 500);
        }

        function nextLevel() {
            modalBox.style.transform = "scale(0.95)";
            setTimeout(() => {
                modalOverlay.classList.add('hidden');
                resetCurrent();
                
                if(xp >= maxXP) {
                    // Reset game if they reached max
                    xp = 0;
                    xpFill.style.width = `0%`;
                    xpDisplay.innerText = `0 / ${maxXP}`;
                }
                
                generateTarget();
            }, 200);
        }

        function fireConfetti() {
            const emojis = ['🍃', '🦒', '✨', '💎', '💛'];
            for (let i = 0; i < 30; i++) {
                const conf = document.createElement('div');
                conf.innerText = emojis[Math.floor(Math.random() * emojis.length)];
                conf.classList.add('confetti');
                conf.style.left = Math.random() * 100 + 'vw';
                conf.style.top = -50 + 'px';
                conf.style.animationDuration = (Math.random() * 2 + 2) + 's';
                document.body.appendChild(conf);
                
                // Cleanup
                setTimeout(() => {
                    conf.remove();
                }, 4000);
            }
        }

        // Start the game
        window.onload = () => {
            // First interaction required for audio API, handled by buttons
            generateTarget();
        };

    </script>
